import { useState } from 'react'
import { Link, useNavigate } from 'react-router-dom'
import {
  LogIn, Phone, Lock, Eye, EyeOff, ArrowLeft,
  AlertTriangle, ShieldCheck, KeyRound,
} from 'lucide-react'

export default function Login() {
  const navigate = useNavigate()
  const [simu, setSimu] = useState('')
  const [password, setPassword] = useState('')
  const [showPassword, setShowPassword] = useState(false)
  const [error, setError] = useState('')

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    if (!simu.trim()) return setError('Namba ya simu inahitajika')
    if (!password.trim()) return setError('Nywila inahitajika')

    // Demo: navigate kulingana na simu
    if (simu.includes('0712') || simu.includes('0754')) {
      navigate('/mteja')
    } else if (simu.includes('0713') || simu.includes('0714')) {
      navigate('/muuzaji')
    } else {
      navigate('/mmiliki')
    }
  }

  return (
    <div className="min-h-screen bg-gradient-to-br from-slate-50 via-white to-brand-50/30 flex items-center justify-center p-4">
      <div className="w-full max-w-md">
        <Link
          to="/"
          className="inline-flex items-center gap-1.5 text-sm text-slate-600 hover:text-brand-600 mb-6"
        >
          <ArrowLeft size={14} />
          Rudi Nyumbani
        </Link>

        <div className="bg-white rounded-2xl border border-slate-200 p-6 md:p-8 shadow-lg">
          <div className="text-center mb-6">
            <div className="w-14 h-14 rounded-2xl bg-brand-50 text-brand-600 grid place-items-center mx-auto">
              <LogIn size={28} />
            </div>
            <h1 className="text-2xl font-bold mt-3">Ingia</h1>
            <p className="text-sm text-slate-500 mt-1">
              Weka namba yako ya simu na nywila
            </p>
          </div>

          <form onSubmit={handleSubmit} className="space-y-3">
            <div>
              <label className="text-sm font-medium">Namba ya Simu</label>
              <div className="relative mt-1">
                <Phone size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
                <input
                  value={simu}
                  onChange={(e) => setSimu(e.target.value)}
                  placeholder="0754-123-456"
                  inputMode="tel"
                  className="w-full pl-10 pr-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
                />
              </div>
            </div>

            <div>
              <label className="text-sm font-medium">Nywila</label>
              <div className="relative mt-1">
                <Lock size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
                <input
                  type={showPassword ? 'text' : 'password'}
                  value={password}
                  onChange={(e) => setPassword(e.target.value)}
                  placeholder="••••••••"
                  className="w-full pl-10 pr-10 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
                />
                <button
                  type="button"
                  onClick={() => setShowPassword(!showPassword)}
                  className="absolute right-2 top-1/2 -translate-y-1/2 p-1.5 rounded-md hover:bg-slate-100"
                >
                  {showPassword ? <EyeOff size={14} /> : <Eye size={14} />}
                </button>
              </div>
            </div>

            {error && (
              <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 text-red-700 text-sm">
                <AlertTriangle size={16} className="mt-0.5 shrink-0" />
                {error}
              </div>
            )}

            <button
              type="submit"
              className="w-full py-2.5 rounded-lg bg-brand-600 text-white text-sm font-semibold hover:bg-brand-700"
            >
              Ingia
            </button>
          </form>

          <div className="mt-4 text-center">
            <Link
              to="/forgot-password"
              className="inline-flex items-center gap-1.5 text-sm text-brand-600 hover:underline"
            >
              <KeyRound size={12} />
              Umesahau Nywila?
            </Link>
          </div>

          <div className="mt-6 pt-4 border-t border-slate-100 text-center">
            <div className="inline-flex items-center gap-1.5 text-xs text-slate-500">
              <ShieldCheck size={12} />
              Usalama wa akaunti yako ni kipaumbele chetu
            </div>
          </div>
        </div>
      </div>
    </div>
  )
},

import { useState } from 'react'
import { Link, useNavigate } from 'react-router-dom'
import {
  KeyRound, Phone, ArrowLeft, AlertTriangle, CheckCircle2,
  ShieldCheck, Smartphone,
} from 'lucide-react'

export default function ForgotPassword() {
  const navigate = useNavigate()
  const [simu, setSimu] = useState('')
  const [otpSent, setOtpSent] = useState(false)
  const [otp, setOtp] = useState('')
  const [error, setError] = useState('')

  const handleSendOtp = (e: React.FormEvent) => {
    e.preventDefault()
    if (!/^0[67]\d{2}-?\d{3}-?\d{3}$/.test(simu.replace(/\s/g, ''))) {
      return setError('Weka namba sahihi ya simu')
    }
    setError('')
    setOtpSent(true)
  }

  const handleVerifyOtp = (e: React.FormEvent) => {
    e.preventDefault()
    if (otp.length !== 6) return setError('OTP inapaswa kuwa tarakimu 6')
    setError('')
    navigate('/reset-password')
  }

  return (
    <div className="min-h-screen bg-gradient-to-br from-slate-50 via-white to-brand-50/30 flex items-center justify-center p-4">
      <div className="w-full max-w-md">
        <Link
          to="/login"
          className="inline-flex items-center gap-1.5 text-sm text-slate-600 hover:text-brand-600 mb-6"
        >
          <ArrowLeft size={14} />
          Rudi Login
        </Link>

        <div className="bg-white rounded-2xl border border-slate-200 p-6 md:p-8 shadow-lg">
          <div className="text-center mb-6">
            <div className="w-14 h-14 rounded-2xl bg-amber-50 text-amber-600 grid place-items-center mx-auto">
              <KeyRound size={28} />
            </div>
            <h1 className="text-2xl font-bold mt-3">Umesahau Nywila?</h1>
            <p className="text-sm text-slate-500 mt-1">
              {otpSent
                ? 'Weka OTP uliyotumiwa kwa simu'
                : 'Weka namba yako ili tukupe OTP'}
            </p>
          </div>

          {!otpSent ? (
            <form onSubmit={handleSendOtp} className="space-y-3">
              <div>
                <label className="text-sm font-medium">Namba ya Simu</label>
                <div className="relative mt-1">
                  <Phone size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
                  <input
                    value={simu}
                    onChange={(e) => setSimu(e.target.value)}
                    placeholder="0754-123-456"
                    inputMode="tel"
                    className="w-full pl-10 pr-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
                  />
                </div>
              </div>

              {error && (
                <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 text-red-700 text-sm">
                  <AlertTriangle size={16} className="mt-0.5 shrink-0" />
                  {error}
                </div>
              )}

              <button
                type="submit"
                className="w-full inline-flex items-center justify-center gap-1.5 py-2.5 rounded-lg bg-brand-600 text-white text-sm font-semibold hover:bg-brand-700"
              >
                <Smartphone size={14} />
                Tuma OTP
              </button>
            </form>
          ) : (
            <form onSubmit={handleVerifyOtp} className="space-y-3">
              <div className="p-3 rounded-lg bg-emerald-50 border border-emerald-200 flex items-start gap-2">
                <CheckCircle2 size={16} className="text-emerald-600 mt-0.5 shrink-0" />
                <div className="text-xs text-emerald-800">
                  OTP imetumwa kwa <span className="font-semibold">{simu}</span>
                </div>
              </div>

              <div>
                <label className="text-sm font-medium">OTP (tarakimu 6)</label>
                <input
                  value={otp}
                  onChange={(e) => setOtp(e.target.value.replace(/\D/g, '').slice(0, 6))}
                  placeholder="123456"
                  inputMode="numeric"
                  maxLength={6}
                  className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm font-mono text-center text-lg tracking-widest"
                />
              </div>

              {error && (
                <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 text-red-700 text-sm">
                  <AlertTriangle size={16} className="mt-0.5 shrink-0" />
                  {error}
                </div>
              )}

              <button
                type="submit"
                className="w-full inline-flex items-center justify-center gap-1.5 py-2.5 rounded-lg bg-brand-600 text-white text-sm font-semibold hover:bg-brand-700"
              >
                <CheckCircle2 size={14} />
                Thibitisha OTP
              </button>

              <button
                type="button"
                onClick={() => setOtpSent(false)}
                className="w-full text-xs text-slate-500 hover:text-brand-600"
              >
                Badilisha namba ya simu
              </button>
            </form>
          )}

          <div className="mt-6 pt-4 border-t border-slate-100 text-center">
            <div className="inline-flex items-center gap-1.5 text-xs text-slate-500">
              <ShieldCheck size={12} />
              OTP ni salama - usiituma kwa mtu mwingine
            </div>
          </div>
        </div>
      </div>
    </div>
  )
},

import { useNavigate } from 'react-router-dom'
import {
  Wallet, TrendingUp, TrendingDown, HandCoins, ChevronRight,
  ShoppingBag, Receipt, Building2, Activity,
} from 'lucide-react'
import { useStore } from '../../store/StoreContext'
import { formatTsh } from '../../utils/faida'

export default function MmilikiDashboard() {
  const navigate = useNavigate()
  const { maduka, mauzo, opex, orders, maombi, faida } = useStore()

  const f = useMemo(() => faida(), [mauzo, opex])

  const stats = useMemo(() => {
    const ordersZinazosubiri = orders.filter((o) => o.hali === 'inasubiri_malipo').length
    const maombiYanasubiri = maombi.filter((m) => m.hali === 'inasubiri').length
    const jumlaDeni = orders
      .filter((o) => o.hali === 'inasubiri_malipo')
      .reduce((s, o) => s + o.jumla, 0)
    return { ordersZinazosubiri, maombiYanasubiri, jumlaDeni }
  }, [orders, maombi])

  const faidaKwaDuka = useMemo(
    () => maduka.map((d) => ({ duka: d, ...faida(d.id) })),
    [maduka, mauzo, opex]
  )

  const ordersZaKaribuni = orders.slice(0, 5)
  const maombiYaKaribuni = maombi.filter((m) => m.hali === 'inasubiri').slice(0, 3)

  return (
    <div className="space-y-5">
      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-3">
        <div>
          <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-slate-900">
            Dashboard ya Mmiliki
          </h1>
          <p className="text-sm text-slate-500 mt-1">
            Muhtasari wa maduka yote — mauzo, faida, zaka na orders
          </p>
        </div>
        <div className="text-xs text-slate-500 bg-white border border-slate-200 rounded-lg px-3 py-2 self-start sm:self-auto">
          {new Date().toLocaleDateString('sw-TZ', { day: '2-digit', month: 'long', year: 'numeric' })}
        </div>
      </div>

      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
        <Stat label="Mauzo Yote" value={formatTsh(f.mauzo)} sub={`Maduka ${maduka.length}`} Icon={ShoppingBag} tone="text-blue-600" />
        <Stat
          label="Faida Safi"
          value={formatTsh(f.faidaSafi)}
          sub={f.hasara ? 'Biashara ina hasara' : 'Baada ya gharama zote'}
          Icon={f.hasara ? TrendingDown : TrendingUp}
          tone={f.hasara ? 'text-red-600' : 'text-emerald-600'}
        />
        <Stat label="Deni la Kukusanya" value={formatTsh(stats.jumlaDeni)} sub={`Orders ${stats.ordersZinazosubiri}`} Icon={Wallet} tone="text-amber-600" />

              <HandCoins size={24} />
            </div>
            <div>
              <div className="text-sm text-slate-500">Kwa Bwana Wiki Hii</div>
              <div className="text-2xl md:text-3xl font-bold text-slate-900 mt-0.5">
                {formatTsh(f.jumlaYaMungu)}
              </div>
              <div className="text-xs text-slate-500 mt-1">
                Zaka {formatTsh(f.zaka)} + Sadaka {formatTsh(f.sadaka)}
              </div>
            </div>
          </div>
          <button
            onClick={() => navigate('/mmiliki/zaka')}
            className="inline-flex items-center justify-center gap-1.5 px-4 py-2.5 bg-brand-600 text-white rounded-lg text-sm font-semibold hover:bg-brand-700 transition self-start sm:self-auto"

            <Building2 size={14} className="inline mr-1" /> Maduka Yote

            <button key={d.id} onClick={() => setSelectedDuka(d.id)} className={`px-3 py-2 rounded-lg text-sm border ${selectedDuka === d.id ? 'bg-brand-600 text-white border-brand-600 font-semibold' : 'bg-white border-slate-200 hover:bg-slate-50'}`}>

          <div><div className="text-xs text-slate-500">Sadaka (10%)</div><div className="text-base font-bold text-slate-900">{formatTsh(f.sadaka)}</div></div>

        <Stat label="COGS" value={formatTsh(f.cogs)} Icon={Package} tone="text-amber-600" />
        <Stat label="Faida Ghafi" value={formatTsh(f.faidaGhafi)} Icon={TrendingUp} tone="text-emerald-600" />
        <Stat label="Opex" value={formatTsh(f.jumlaOpex)} Icon={Wallet} tone="text-orange-600" sub="Bila mishahara" />
        <Stat label="Mishahara" value={formatTsh(f.jumlaMishahara)} Icon={Users} tone="text-purple-600" />

function Stat({ label, value, sub, Icon, tone }: { label: string; value: string; sub?: string; Icon: React.ElementType; tone: string }) {
  return (
    <div className="bg-white rounded-2xl border border-slate-200 p-4">
      <div className={`w-10 h-10 rounded-lg bg-slate-50 border border-slate-200 grid place-items-center ${tone}`}>
        <Icon size={18} />
      </div>
      <div className="text-xs text-slate-500 mt-3">{label}</div>
      <div className="text-base md:text-lg font-bold text-slate-900 mt-0.5 truncate">{value}</div>
      {sub && <div className="text-[10px] text-slate-500">{sub}</div>}
    </div>
  )
}

function Row({ label, value, tone, bold }: { label: string; value: string; tone?: string; bold?: boolean }) {
  return (
    <div className="flex justify-between">
      <span className="text-slate-600">{label}</span>
      <span className={`${bold ? 'font-bold' : 'font-semibold'} ${tone ?? 'text-slate-900'}`}>{value}</span>
    </div>
  )
},
import { useMemo, useState } from 'react'
import { Wallet, Banknote, CheckCircle2, Clock, Users, Zap } from 'lucide-react'
import { useStore } from '../../store/StoreContext'
import { formatTsh } from '../../utils/faida'

export default function MmilikiMishahara() {
  const { mishahara, lipaMshahara } = useStore()
  const [filterHali, setFilterHali] = useState<'zote' | 'inasubiri' | 'imelipwa'>('zote')

  const filtered = useMemo(() => {
    return mishahara.filter((m) => filterHali === 'zote' || m.hali === filterHali)
  }, [mishahara, filterHali])

  const stats = useMemo(() => {
    const zote = mishahara.length
    const jumla = mishahara.reduce((s, m) => s + m.jumla, 0)
    const zilizolipwa = mishahara.filter((m) => m.hali === 'imelipwa')
    const jumlaIliyolipwa = zilizolipwa.reduce((s, m) => s + m.jumla, 0)
    const inasubiri = mishahara.filter((m) => m.hali === 'inasubiri')
    const jumlaInasubiri = inasubiri.reduce((s, m) => s + m.jumla, 0)
    return { zote, jumla, zilizolipwa: zilizolipwa.length, jumlaIliyolipwa, inasubiri: inasubiri.length, jumlaInasubiri }
  }, [mishahara])

  const lipaZote = () => {
    mishahara.filter((m) => m.hali === 'inasubiri').forEach((m) => lipaMshahara(m.id))
  }

  return (
    <div className="space-y-5">
      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-3">
        <div>
          <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-slate-900 flex items-center gap-2">
            <Wallet size={26} className="text-brand-600" /> Usimamizi wa Mishahara
          </h1>
          <p className="text-sm text-slate-500 mt-1">Mishahara inaingia moja kwa moja kwenye gharama za biashara</p>
        </div>
        {stats.inasubiri > 0 && (
          <button onClick={lipaZote} className="inline-flex items-center gap-1.5 px-4 py-2.5 bg-emerald-600 text-white rounded-lg text-sm font-semibold hover:bg-emerald-700 self-start sm:self-auto">
            <Zap size={16} /> Lipa Wote ({stats.inasubiri})
          </button>
        )}
      </div>

      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
        <Stat label="Mishahara Yote" value={String(stats.zote)} Icon={Users} tone="text-blue-600" />
        <Stat label="Jumla" value={formatTsh(stats.jumla)} Icon={Wallet} tone="text-purple-600" />
        <Stat label="Zilizolipwa" value={formatTsh(stats.jumlaIliyolipwa)} Icon={CheckCircle2} tone="text-emerald-600" sub={`${stats.zilizolipwa} wafanyakazi`} />
        <Stat label="Zinasubiri" value={formatTsh(stats.jumlaInasubiri)} Icon={Clock} tone="text-amber-600" sub={`${stats.inasubiri} wafanyakazi`} />
      </div>

      <div className="bg-white rounded-2xl border border-slate-200 p-3 md:p-4 flex flex-wrap gap-2">
        {(['zote', 'inasubiri', 'imelipwa'] as const).map((k) => (
          <button key={k} onClick={() => setFilterHali(k)} className={`px-3 py-1.5 rounded-lg text-xs border ${filterHali === k ? 'bg-brand-600 text-white border-brand-600' : 'bg-white border-slate-200 hover:bg-slate-50'}`}>
            {k === 'zote' ? 'Zote' : k === 'inasubiri' ? 'Zinasubiri' : 'Zilizolipwa'}
          </button>
        ))}
      </div>

      {filtered.length === 0 ? (
        <div className="bg-white rounded-2xl border border-slate-200 p-12 text-center">
          <Wallet size={40} className="mx-auto text-slate-300 mb-3" />
          <h3 className="font-semibold">Hakuna mishahara</h3>
        </div>
      ) : (
        <div className="space-y-2">
          {filtered.map((m) => (
            <div key={m.id} className="bg-white rounded-2xl border border-slate-200 p-4">
              <div className="flex flex-col sm:flex-row sm:justify-between gap-3">
                <div className="flex items-center gap-3 min-w-0">
                  <div className="w-10 h-10 rounded-full bg-slate-50 border border-slate-200 text-slate-700 grid place-items-center text-xs font-bold shrink-0">
                    {m.jinaMtumiaji.split(' ').map((n) => n[0]).slice(0, 2).join('')}
                  </div>
                  <div className="min-w-0">
                    <div className="font-semibold text-sm">{m.jinaMtumiaji}</div>
                    <div className="text-xs text-slate-500">{m.nafasi} — {m.dukaJina.replace('Smart Hardware - ', '')}</div>
                    <div className="text-xs text-slate-500">{m.mwezi}</div>
                  </div>
                </div>
                <div className="sm:text-right shrink-0">
                  <div className="text-base font-bold text-slate-900">{formatTsh(m.jumla)}</div>
                  <div className={`text-[10px] uppercase font-semibold ${m.hali === 'imelipwa' ? 'text-emerald-600' : 'text-amber-600'}`}>
                    {m.hali === 'imelipwa' ? 'Imelipwa' : 'Inasubiri'}
                  </div>
                </div>
              </div>
              {m.hali === 'inasubiri' && (
                <div className="mt-3 pt-3 border-t border-slate-100 flex gap-2">
                  <button onClick={() => lipaMshahara(m.id)} className="inline-flex items-center gap-1 px-3 py-1.5 rounded-md bg-emerald-600 text-white text-xs font-semibold hover:bg-emerald-700">
                    <Banknote size={12} /> Lipa
                  </button>
                </div>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  )
}

        <Icon size={18} />

      {sub && <div className="text-[10px] text-slate-500">{sub}</div>}
    </div>
  )
},
import { useMemo, useState } from 'react'
import {
  ShoppingBag, Search, Filter, CheckCircle2, Clock, XCircle,
  X, Package, Phone, CreditCard, AlertTriangle, Loader2, Check, Ban,
} from 'lucide-react'
import { useStore } from '../../store/StoreContext'
import { formatTsh } from '../../utils/faida'
import type { Order } from '../../types'

const HALI_META: Record<
  Order['hali'],
  { label: string; tone: string; Icon: React.ElementType }
> = {
  inasubiri_malipo: { label: 'Inasubiri Malipo', tone: 'text-amber-700', Icon: Clock },
  imekamilika: { label: 'Imekamilika', tone: 'text-emerald-700', Icon: CheckCircle2 },
  imekataliwa: { label: 'Imekataliwa', tone: 'text-red-700', Icon: XCircle },
}

const MALIPO_HALI_META: Record<string, { label: string; tone: string; bg: string; border: string; Icon: React.ElementType }> = {
  inasubiri: {
    label: 'Inasubiri Uthibitisho',
    tone: 'text-amber-700',
    bg: 'bg-amber-50',
    border: 'border-amber-200',
    Icon: Clock,
  },
  imethibitishwa: {
    label: 'Imethibitishwa',
    tone: 'text-emerald-700',
    bg: 'bg-emerald-50',
    border: 'border-emerald-200',
    Icon: CheckCircle2,
  },
  imekataliwa: {
    label: 'Imekataliwa',
    tone: 'text-red-700',
    bg: 'bg-red-50',
    border: 'border-red-200',
    Icon: XCircle,
  },
}

export default function MmilikiOrders() {
  const { orders, thibitishaMalipo, kataaMalipo } = useStore()
  const [q, setQ] = useState('')
  const [filter, setFilter] = useState<Order['hali'] | 'zote'>('zote')
  const [selected, setSelected] = useState<Order | null>(null)
  const [processing, setProcessing] = useState<string | null>(null)
  const [toast, setToast] = useState<{ msg: string; ok: boolean } | null>(null)

  const filtered = useMemo(() => {
    const query = q.toLowerCase().trim()
    return orders.filter((o) => {
      if (filter !== 'zote' && o.hali !== filter) return false
      if (!query) return true
      return (
        o.id.toLowerCase().includes(query) ||
        o.mteja.toLowerCase().includes(query) ||
        o.simu.includes(query)
      )
    })
  }, [orders, q, filter])

  const stats = useMemo(() => {
    const zote = orders.length
    const zinazosubiri = orders.filter((o) => o.hali === 'inasubiri_malipo').length
    const zilizokamilika = orders.filter((o) => o.hali === 'imekamilika').length
    const jumlaMapato = orders
      .filter((o) => o.hali === 'imekamilika')
      .reduce((s, o) => s + o.jumla, 0)
    const jumlaInayosubiri = orders
      .filter((o) => o.hali === 'inasubiri_malipo')
      .reduce((s, o) => s + o.jumla, 0)
    const malipoYanasubiri = orders.reduce(
      (s, o) => s + o.malipo.filter((m) => m.hali === 'inasubiri').length,
      0
    )
    return {
      zote,
      zinazosubiri,
      zilizokamilika,
      jumlaMapato,
      jumlaInayosubiri,
      malipoYanasubiri,
    }
  }, [orders])

  // ============================================================
  // Thibitisha malipo
  // ============================================================
  const handleThibitisha = async (orderId: string, malipoId: string) => {
    setProcessing(malipoId)
    await new Promise((r) => setTimeout(r, 400))
    thibitishaMalipo(orderId, malipoId)
    setProcessing(null)
    setToast({ msg: 'Malipo yamethibitishwa. Order imekamilika.', ok: true })
    setTimeout(() => setToast(null), 4000)

    // Sasisha order iliyo kwenye modal
    const updated = orders.find((o) => o.id === orderId)
    if (updated && selected?.id === orderId) {
      const jumlaIlipwa = updated.malipo
        .filter((m) => m.hali === 'imethibitishwa')
        .reduce((s, m) => s + m.kiasi, 0)
      setSelected({
        ...updated,
        hali: jumlaIlipwa >= updated.jumla ? 'imekamilika' : updated.hali,
      })
    }
  }

  // ============================================================
  // Kataa malipo
  // ============================================================
  const handleKataa = async (orderId: string, malipoId: string) => {
    if (!confirm('Una uhakika unataka kukataa malipo haya?')) return
    setProcessing(malipoId)
    await new Promise((r) => setTimeout(r, 400))
    kataaMalipo(orderId, malipoId)
    setProcessing(null)
    setToast({ msg: 'Malipo yamekataliwa.', ok: false })
    setTimeout(() => setToast(null), 4000)
  }

  return (
    <div className="space-y-5">
      <div>
        <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-slate-900 flex items-center gap-2">
          <ShoppingBag size={26} className="text-brand-600" />
          Orders Zote
        </h1>
        <p className="text-sm text-slate-500 mt-1">
          Simamia orders na uthibitishe malipo ya mobile
        </p>
      </div>

      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
        <Stat label="Orders Zote" value={String(stats.zote)} Icon={ShoppingBag} tone="text-blue-600" />
        <Stat
          label="Zinasubiri"
          value={String(stats.zinazosubiri)}
          Icon={Clock}
          tone="text-amber-600"
          sub={formatTsh(stats.jumlaInayosubiri)}
        />
        <Stat
          label="Zilizokamilika"
          value={String(stats.zilizokamilika)}
          Icon={CheckCircle2}
          tone="text-emerald-600"
        />
        <Stat
          label="Malipo Yanasubiri"
          value={String(stats.malipoYanasubiri)}
          Icon={CreditCard}
          tone={stats.malipoYanasubiri > 0 ? 'text-amber-600' : 'text-slate-500'}
        />
      </div>

      {/* Filter */}
      <div className="bg-white rounded-2xl border border-slate-200 p-3 md:p-4 space-y-3">
        <div className="relative">
          <Search size={18} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-slate-400" />
          <input
            value={q}
            onChange={(e) => setQ(e.target.value)}
            placeholder="Tafuta kwa order ID, mteja, au simu..."
            className="w-full pl-11 pr-4 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
        <div className="flex flex-wrap gap-2 items-center">
          <Filter size={14} className="text-slate-400" />
          {(['zote', 'inasubiri_malipo', 'imekamilika', 'imekataliwa'] as const).map((k) => (
            <button
              key={k}
              onClick={() => setFilter(k)}
              className={`px-3 py-1.5 rounded-lg text-xs border transition ${
                filter === k
                  ? 'bg-brand-600 text-white border-brand-600 font-semibold'
                  : 'bg-white border-slate-200 hover:bg-slate-50'
              }`}
            >
              {k === 'zote' ? 'Zote' : HALI_META[k].label}
            </button>
          ))}
        </div>
      </div>

      {/* Orders List */}
      {filtered.length === 0 ? (
        <div className="bg-white rounded-2xl border border-slate-200 p-12 text-center">
          <ShoppingBag size={40} className="mx-auto text-slate-300 mb-3" />
          <h3 className="font-semibold">Hakuna orders</h3>
          <p className="text-sm text-slate-500 mt-1">Badilisha kichujio au utafutaji.</p>
        </div>
      ) : (
        <div className="space-y-3">
          {filtered.map((o) => {
            const meta = HALI_META[o.hali]
            const jumlaIlipwa = o.malipo
              .filter((m) => m.hali === 'imethibitishwa')
              .reduce((s, m) => s + m.kiasi, 0)
            const baki = o.jumla - jumlaIlipwa
            const malipoYanasubiri = o.malipo.filter((m) => m.hali === 'inasubiri').length

            return (
              <button
                key={o.id}
                onClick={() => setSelected(o)}

                        <span className={`text-[10px] uppercase font-semibold ${meta.tone}`}>

                        </span>
                        {malipoYanasubiri > 0 && (
                          <span className="inline-flex items-center gap-1 text-[10px] font-semibold text-amber-700 bg-amber-50 border border-amber-200 px-1.5 py-0.5 rounded">

                      </div>
                      <div className="text-xs text-slate-500">
                        {o.tarehe} — {o.bidhaa.length} bidhaa
                      </div>
                    </div>
                  </div>
                  <div className="sm:text-right shrink-0">
                    <div className="text-base font-bold text-slate-900">
                      {formatTsh(o.jumla)}
                    </div>
                    {baki > 0 ? (
                      <div className="text-xs text-amber-700">
                        Baki: {formatTsh(baki)}
                      </div>
                    ) : (
                      <div className="text-xs text-emerald-700 font-semibold">
                        Imelipwa Yote
                      </div>
                    )}
                  </div>
                </div>
              </button>
            )
          })}
        </div>
      )}

      {selected && (
        <OrderDetail
          order={selected}
          onClose={() => setSelected(null)}
          onThibitisha={handleThibitisha}
          onKataa={handleKataa}
          processing={processing}
        />
      )}

      {toast && (
        <div
          className={`fixed bottom-4 left-1/2 -translate-x-1/2 z-[60] text-white text-sm px-4 py-3 rounded-xl shadow-2xl flex items-center gap-2 max-w-md ${
            toast.ok ? 'bg-emerald-600' : 'bg-red-600'
          }`}
        >
          {toast.ok ? <CheckCircle2 size={16} /> : <XCircle size={16} />}
          {toast.msg}
        </div>
      )}
    </div>
  )
}

// Picha Upload — hakuna size limit, compression ya lazima, AlertModal
// ============================================================
function PichaUpload({
  value, onChange,
}: {
  value?: string
  onChange: (dataUrl: string | undefined) => void
}) {
  const inputRef = useRef<HTMLInputElement>(null)
  const [alert, setAlert] = useState<{ title: string; message: string; tone: 'danger' | 'warning' } | null>(null)

  const compressAndSet = (file: File) => {
    const reader = new FileReader()
    reader.onload = () => {
      const img = new Image()
      img.onload = () => {
        // Punguza hadi max 800x800 — inatosha kwa bidhaa
        const MAX = 800
        let { width, height } = img
        if (width > height && width > MAX) {
          height = Math.round((height * MAX) / width)
          width = MAX
        } else if (height > MAX) {
          width = Math.round((width * MAX) / height)
          height = MAX
        }

        const canvas = document.createElement('canvas')
        canvas.width = width
        canvas.height = height
        const ctx = canvas.getContext('2d')
        if (!ctx) {
          setAlert({
            title: 'Kifaa hakiwezi kuchakata picha',
            message: 'Kivinjari chako hakina uwezo wa kuchakata picha. Jaribu kivinjari kingine.',
            tone: 'danger',
          })
          return
        }
        ctx.drawImage(img, 0, 0, width, height)

        // Anza na quality ya juu, punguza kama ni kubwa sana
        let dataUrl = canvas.toDataURL('image/jpeg', 0.85)
        let quality = 0.85
        while (dataUrl.length > 800_000 && quality > 0.3) {
          quality -= 0.1
          dataUrl = canvas.toDataURL('image/jpeg', quality)
        }

        onChange(dataUrl)
      }
      img.onerror = () => {
        setAlert({
          title: 'Imeshindwa kusoma picha',
          message: 'Faili uliyopakia halifai kama picha. Jaribu picha nyingine (PNG, JPG, WEBP).',
          tone: 'danger',
        })
      }
      img.src = reader.result as string
    }
    reader.onerror = () => {
      setAlert({
        title: 'Imeshindwa kusoma faili',
        message: 'Faili halikusomwa. Jaribu tena au chagua faili lingine.',
        tone: 'danger',
      })
    }
    reader.readAsDataURL(file)
  }

  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return
    if (!file.type.startsWith('image/')) {
      setAlert({
        title: 'Faili si picha',
        message: 'Tafadhali chagua picha halali (PNG, JPG, WEBP, GIF, n.k.).',
        tone: 'warning',
      })
      return
    }
    compressAndSet(file)
  }

  return (
    <div>
      <label className="text-sm font-medium">Picha ya Bidhaa</label>
      <div className="mt-1.5 flex items-center gap-3">
        <div className="w-20 h-20 rounded-xl bg-slate-50 border border-slate-200 grid place-items-center overflow-hidden shrink-0">
          {value ? (
            <img src={value} alt="picha" className="w-full h-full object-cover" />
          ) : (
            <ImageIcon size={24} className="text-slate-300" />
          )}
        </div>
        <div className="flex-1 min-w-0 space-y-1.5">
          <input
            ref={inputRef}
            type="file"
            accept="image/*"
            onChange={handleFile}
            className="hidden"
          />
          <div className="flex flex-wrap items-center gap-2">
            <button
              type="button"
              onClick={() => inputRef.current?.click()}
              className="inline-flex items-center gap-1.5 px-3 py-2 rounded-lg bg-slate-100 hover:bg-slate-200 text-sm font-medium"
            >
              <Upload size={14} /> {value ? 'Badilisha Picha' : 'Pakia Picha'}
            </button>
            {value && (
              <button
                type="button"
                onClick={() => onChange(undefined)}
                className="inline-flex items-center gap-1 px-2 py-1 rounded-md text-xs text-red-600 hover:bg-red-50"
              >
                <X size={12} /> Ondoa
              </button>
            )}
          </div>
          <div className="text-[10px] text-slate-500 flex items-center gap-1">
            <AlertTriangle size={10} className="text-amber-500" />
            Picha yoyote — itapunguzwa hadi 800px automatically
          </div>
        </div>
      </div>

      <AlertModal
        open={!!alert}
        title={alert?.title ?? ''}
        message={alert?.message ?? ''}
        tone={alert?.tone ?? 'danger'}
        onClose={() => setAlert(null)}
      />
    </div>
  )
}

  dukaId, dukaJina, mtumiaji, nafasi,

  dukaId: string
  dukaJina: string
  mtumiaji: string
  nafasi: 'mmiliki' | 'admin'
}) {
  const { ongezaUingizaji, maduka } = useStore()
  const [jina, setJina] = useState('')
  const [sku, setSku] = useState('')
  const [skuManual, setSkuManual] = useState(false)
  const [kundi, setKundi] = useState<KundiBidhaa>('Saruji')
  const [beiMauzo, setBeiMauzo] = useState(0)
  const [beiUnunuzi, setBeiUnunuzi] = useState(0)
  const [uom, setUom] = useState('Mfuko')
  const [stoki, setStoki] = useState(0)
  const [location, setLocation] = useState(dukaJina)
  const [expiry, setExpiry] = useState('')
  const [maelezo, setMaelezo] = useState('')
  const [pichaUrl, setPichaUrl] = useState<string | undefined>()
  const [error, setError] = useState('')
  const [success, setSuccess] = useState('')

  const handleJinaChange = (value: string) => {
    setJina(value)
    if (!skuManual && value.trim()) {
      setSku(generateSku(value, kundi))
    }
  }

  const handleKundiChange = (value: KundiBidhaa) => {
    setKundi(value)
    if (!skuManual && jina.trim()) {
      setSku(generateSku(jina, value))
    }
  }

  const regenerateSku = () => {
    if (jina.trim()) {
      setSku(generateSku(jina, kundi))
      setSkuManual(false)
    }
  }

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    if (!jina.trim()) return setError('Jina la bidhaa linahitajika')
    if (!sku.trim()) return setError('SKU inahitajika')
    if (beiMauzo <= 0) return setError('Bei ya mauzo lazima iwe zaidi ya sifuri')
    if (beiUnunuzi <= 0) return setError('Bei ya ununuzi lazima iwe zaidi ya sifuri')
    if (stoki <= 0) return setError('Stoki lazima iwe zaidi ya sifuri')
    if (!location.trim()) return setError('Chagua mahali (duka)')

    setError('')
    const bidhaa: BidhaaMpya = {
      jina: jina.trim(),
      sku: sku.trim().toUpperCase(),
      kundi,
      beiMauzo,
      beiUnunuzi,
      uom,
      stoki,
      location: location.trim(),
      expiry: expiry || undefined,
      iconKey: kundi,
      pichaUrl,
    }
    const u = ongezaUingizaji({
      aina: 'moja',
      dukaId,
      dukaJina,
      bidhaa: [bidhaa],
      aliyeingiza: mtumiaji,
      nafasi,
      maelezo,
    })
    setSuccess(`Bidhaa imeingizwa! ID: ${u.id} — Muuzaji atathibitisha "Nimepokea"`)
    // Reset
    setJina('')
    setSku('')
    setSkuManual(false)
    setBeiMauzo(0)
    setBeiUnunuzi(0)
    setStoki(0)
    setLocation(dukaJina)
    setExpiry('')
    setMaelezo('')
    setPichaUrl(undefined)
    setTimeout(() => setSuccess(''), 5000)
  }

  const Icon = iconForKundi(kundi)

  return (
    <form onSubmit={handleSubmit} className="bg-white rounded-2xl border border-slate-200 p-4 md:p-5 space-y-4">
      <div className="flex items-center gap-2 text-sm text-slate-600">
        <Store size={14} /> Duka: <span className="font-semibold text-slate-800">{dukaJina}</span>
      </div>

      {success && (
        <div className="flex items-start gap-2 p-3 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-800 text-sm">
          <CheckCircle2 size={16} className="mt-0.5 shrink-0" />
          {success}
        </div>
      )}

      {/* Picha */}
      <PichaUpload value={pichaUrl} onChange={setPichaUrl} />

      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
        <div className="md:col-span-2">
          <label className="text-sm font-medium">Jina la Bidhaa</label>
          <input
            value={jina}
            onChange={(e) => handleJinaChange(e.target.value)}
            placeholder="Mfano: Saruji Dangote 50kg"
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>

        <div>
          <div className="flex items-center justify-between">
            <label className="text-sm font-medium">SKU</label>
            {jina.trim() && (
              <button
                type="button"
                onClick={regenerateSku}
                className="text-[10px] text-brand-600 hover:underline font-semibold"
              >
                Auto-tengeneza
              </button>
            )}
          </div>
          <input
            value={sku}
            onChange={(e) => { setSku(e.target.value.toUpperCase()); setSkuManual(true) }}
            placeholder={jina.trim() ? generateSku(jina, kundi) : 'SAR-DAN-1234'}
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm font-mono"
          />
          <div className="text-[10px] text-slate-500 mt-1">
            {skuManual ? 'Umeandika mwenyewe' : 'Auto-generated kutokana na jina'}
          </div>
        </div>

        <div>
          <label className="text-sm font-medium">Kundi</label>
          <select
            value={kundi}
            onChange={(e) => handleKundiChange(e.target.value as KundiBidhaa)}
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm bg-white"
          >
            {KUNDI.map((k) => <option key={k} value={k}>{k}</option>)}
          </select>
        </div>

        <div>
          <label className="text-sm font-medium">Bei ya Mauzo (Tsh)</label>
          <input
            type="number"
            value={beiMauzo || ''}
            onChange={(e) => setBeiMauzo(Number(e.target.value))}
            placeholder="18500"
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
        <div>
          <label className="text-sm font-medium">Bei ya Ununuzi (Tsh)</label>
          <input
            type="number"
            value={beiUnunuzi || ''}
            onChange={(e) => setBeiUnunuzi(Number(e.target.value))}
            placeholder="16000"
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
        <div>
          <label className="text-sm font-medium">Kipimo (UoM)</label>
          <select
            value={uom}
            onChange={(e) => setUom(e.target.value)}
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm bg-white"
          >
            {UOM.map((u) => <option key={u} value={u}>{u}</option>)}
          </select>
        </div>
        <div>
          <label className="text-sm font-medium">Stoki</label>
          <input
            type="number"
            value={stoki || ''}
            onChange={(e) => setStoki(Number(e.target.value))}
            placeholder="200"
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>

        {/* MAHALI = SELECT YA MADUKA */}
        <div>
          <label className="text-sm font-medium">Mahali (Duka)</label>
          <select
            value={location}
            onChange={(e) => setLocation(e.target.value)}
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm bg-white"
          >
            <option value="">— Chagua duka —</option>
            {maduka.map((d) => (
              <option key={d.id} value={d.jina}>
                {d.jina} — {d.mahali}
              </option>
            ))}
          </select>
        </div>

        <div>
          <label className="text-sm font-medium">Tarehe ya Kuisha (hiari)</label>
          <input
            type="date"
            value={expiry}
            onChange={(e) => setExpiry(e.target.value)}
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
        <div className="md:col-span-2">
          <label className="text-sm font-medium">Maelezo (hiari)</label>
          <input
            value={maelezo}
            onChange={(e) => setMaelezo(e.target.value)}
            placeholder="Mfano: Saruji mpya kutoka Dangote"
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
      </div>

      {/* Preview */}
      <div className="p-3 rounded-lg bg-slate-50 border border-slate-100 flex items-center gap-3">
        <div className="w-14 h-14 rounded-xl bg-white border border-slate-200 overflow-hidden grid place-items-center text-slate-600 shrink-0">
          {pichaUrl ? (
            <img src={pichaUrl} alt="" className="w-full h-full object-cover" />
          ) : (
            <Icon size={22} />
          )}
        </div>
        <div className="flex-1 min-w-0">
          <div className="text-sm font-medium truncate">{jina || 'Jina la bidhaa'}</div>
          <div className="text-xs text-slate-500 truncate">
            {sku || 'SKU auto'} — {kundi} — {stoki} {uom}
          </div>
          <div className="text-xs text-slate-500 truncate">
            {location || 'Duka halijachaguliwa'} — Tsh {beiMauzo.toLocaleString()} / {uom}
          </div>
        </div>
        <div className="text-sm font-bold text-brand-700 whitespace-nowrap">
          Tsh {(stoki * beiUnunuzi).toLocaleString()}
        </div>
      </div>

      {error && (
        <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 text-red-700 text-sm">
          <AlertTriangle size={16} className="mt-0.5 shrink-0" /> {error}
        </div>
      )}

      <button
        type="submit"
        className="w-full py-2.5 rounded-lg bg-brand-600 text-white text-sm font-semibold hover:bg-brand-700 inline-flex items-center justify-center gap-2"
      >
        <Save size={16} /> Ingiza Bidhaa
      </button>
    </form>
  )
}

// ============================================================
// Bulk (Wengi kwa mkupuo)
// ============================================================
interface BulkRow extends BidhaaMpya {
  _id: string
  _skuManual?: boolean
}

function FomuBulk({
  dukaId, dukaJina, mtumiaji, nafasi, maduka,
}: {
  dukaId: string
  dukaJina: string
  mtumiaji: string
  nafasi: 'mmiliki' | 'admin'
  maduka: Duka[]
}) {
  const { ongezaUingizaji } = useStore()
  const [rows, setRows] = useState<BulkRow[]>([emptyRow(dukaJina)])
  const [maelezo, setMaelezo] = useState('')
  const [error, setError] = useState('')
  const [success, setSuccess] = useState('')

  const jumlaBidhaa = rows.reduce((s, r) => s + (r.stoki || 0), 0)
  const jumlaThamani = rows.reduce((s, r) => s + (r.stoki || 0) * (r.beiUnunuzi || 0), 0)

  const updateRow = (id: string, patch: Partial<BulkRow>) => {
    setRows((prev) => prev.map((r) => (r._id === id ? { ...r, ...patch } : r)))
  }

  const addRow = () => setRows((prev) => [...prev, emptyRow(dukaJina)])
  const removeRow = (id: string) => setRows((prev) => prev.filter((r) => r._id !== id))

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    if (rows.length === 0) return setError('Ongeza angalau bidhaa moja')
    for (let i = 0; i < rows.length; i++) {
      const r = rows[i]
      if (!r.jina.trim()) return setError(`Mstari ${i + 1}: Jina linahitajika`)
      if (!r.sku.trim()) return setError(`Mstari ${i + 1}: SKU inahitajika`)
      if (r.beiMauzo <= 0) return setError(`Mstari ${i + 1}: Bei ya mauzo si sahihi`)
      if (r.beiUnunuzi <= 0) return setError(`Mstari ${i + 1}: Bei ya ununuzi si sahihi`)
      if (r.stoki <= 0) return setError(`Mstari ${i + 1}: Stoki si sahihi`)
      if (!r.location.trim()) return setError(`Mstari ${i + 1}: Chagua mahali (duka)`)
    }

    setError('')
    const bidhaa: BidhaaMpya[] = rows.map((r) => ({
      jina: r.jina.trim(),
      sku: r.sku.trim().toUpperCase(),
      kundi: r.kundi,
      beiMauzo: r.beiMauzo,
      beiUnunuzi: r.beiUnunuzi,
      uom: r.uom,
      stoki: r.stoki,
      location: r.location.trim(),
      expiry: r.expiry || undefined,
      iconKey: r.kundi,
      pichaUrl: r.pichaUrl,
    }))

    const u = ongezaUingizaji({
      aina: 'bulk',
      dukaId,
      dukaJina,
      bidhaa,
      aliyeingiza: mtumiaji,
      nafasi,
      maelezo,
    })
    setSuccess(`${bidhaa.length} bidhaa zimeingizwa! ID: ${u.id} — Muuzaji atathibitisha "Nimepokea"`)
    setRows([emptyRow(dukaJina)])
    setMaelezo('')
    setTimeout(() => setSuccess(''), 5000)
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-3">
      <div className="flex items-center gap-2 text-sm text-slate-600 bg-white rounded-2xl border border-slate-200 p-3">
        <Store size={14} /> Duka: <span className="font-semibold text-slate-800">{dukaJina}</span>
      </div>

      {success && (
        <div className="flex items-start gap-2 p-3 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-800 text-sm">
          <CheckCircle2 size={16} className="mt-0.5 shrink-0" />
          {success}
        </div>
      )}

      <div className="bg-white rounded-2xl border border-slate-200 p-4 md:p-5 space-y-3">
        <div className="flex items-center justify-between">
          <h3 className="font-semibold text-sm">Bidhaa ({rows.length})</h3>
          <button
            type="button"
            onClick={addRow}
            className="inline-flex items-center gap-1 px-2.5 py-1.5 rounded-lg bg-brand-50 text-brand-700 text-xs font-semibold hover:bg-brand-100"
          >
            <Plus size={12} /> Ongeza Mstari
          </button>
        </div>

        <div className="space-y-2">
          {rows.map((r, idx) => (
            <BulkRowItem
              key={r._id}
              index={idx}
              row={r}
              maduka={maduka}
              onChange={(patch) => updateRow(r._id, patch)}
              onRemove={() => removeRow(r._id)}
              disabled={rows.length === 1}
            />
          ))}
        </div>

        <div>
          <label className="text-sm font-medium">Maelezo ya Jumla (hiari)</label>
          <input
            value={maelezo}
            onChange={(e) => setMaelezo(e.target.value)}
            placeholder="Mfano: Mizigo ya Novemba kutoka msambazaji"
            className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>

        <div className="p-3 rounded-lg bg-slate-50 border border-slate-100 grid grid-cols-2 gap-2 text-sm">
          <div>
            <div className="text-xs text-slate-500">Jumla Bidhaa</div>
            <div className="font-bold">{jumlaBidhaa} pcs</div>
          </div>
          <div className="text-right">
            <div className="text-xs text-slate-500">Thamani (Ununuzi)</div>
            <div className="font-bold text-brand-700">Tsh {jumlaThamani.toLocaleString()}</div>
          </div>
        </div>

        {error && (
          <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 text-red-700 text-sm">
            <AlertTriangle size={16} className="mt-0.5 shrink-0" /> {error}
          </div>
        )}

        <button
          type="submit"
          className="w-full py-2.5 rounded-lg bg-brand-600 text-white text-sm font-semibold hover:bg-brand-700 inline-flex items-center justify-center gap-2"
        >
          <Save size={16} /> Ingiza Bidhaa Zote ({rows.length})
        </button>
      </div>
    </form>
  )
}

function BulkRowItem({
  index, row, maduka, onChange, onRemove, disabled,
}: {
  index: number
  row: BulkRow
  maduka: Duka[]
  onChange: (patch: Partial<BulkRow>) => void
  onRemove: () => void
  disabled: boolean
}) {
  const [open, setOpen] = useState(true)
  const Icon = iconForKundi(row.kundi)

  const handleJina = (value: string) => {
    const patch: Partial<BulkRow> = { jina: value }
    if (!row._skuManual && value.trim()) {
      patch.sku = generateSku(value, row.kundi)
    }
    onChange(patch)
  }

  const handleKundi = (value: KundiBidhaa) => {
    const patch: Partial<BulkRow> = { kundi: value, iconKey: value }
    if (!row._skuManual && row.jina.trim()) {
      patch.sku = generateSku(row.jina, value)
    }
    onChange(patch)
  }

  const handleSku = (value: string) => {
    onChange({ sku: value.toUpperCase(), _skuManual: true })
  }

  return (
    <div className="border border-slate-200 rounded-lg overflow-hidden">
      <div className="flex items-center gap-2 p-2 bg-slate-50">
        <button
          type="button"
          onClick={() => setOpen((v) => !v)}
          className="p-1 rounded-md hover:bg-slate-200"
        >
          <ChevronDown size={14} className={`transition ${open ? '' : '-rotate-90'}`} />
        </button>
        <div className="w-7 h-7 rounded-md bg-white border border-slate-200 overflow-hidden grid place-items-center text-slate-600 shrink-0">
          {row.pichaUrl ? (
            <img src={row.pichaUrl} alt="" className="w-full h-full object-cover" />
          ) : (
            <Icon size={12} />
          )}
        </div>
        <div className="flex-1 min-w-0 text-xs">
          <div className="font-medium truncate">
            {row.jina || `Bidhaa #${index + 1}`}
          </div>
          <div className="text-slate-500 truncate">
            {row.sku || 'SKU auto'} — {row.stoki} {row.uom} — {row.location || 'Duka'} — Tsh {row.beiMauzo.toLocaleString()}
          </div>
        </div>
        <button
          type="button"
          onClick={onRemove}
          disabled={disabled}
          className="p-1.5 rounded-md text-red-500 hover:bg-red-50 disabled:opacity-40 disabled:cursor-not-allowed"
        >
          <Trash2 size={12} />
        </button>
      </div>

      {open && (
        <div className="p-3 space-y-2">
          <PichaUpload value={row.pichaUrl} onChange={(url) => onChange({ pichaUrl: url })} />

          <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
            <div className="col-span-2 md:col-span-4">
              <input
                value={row.jina}
                onChange={(e) => handleJina(e.target.value)}
                placeholder="Jina la bidhaa"
                className="w-full px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none"
              />
            </div>
            <input
              value={row.sku}
              onChange={(e) => handleSku(e.target.value)}
              placeholder="SKU auto"
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs font-mono focus:border-brand-500 outline-none"
            />
            <select
              value={row.kundi}
              onChange={(e) => handleKundi(e.target.value as KundiBidhaa)}
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none bg-white"
            >
              {KUNDI.map((k) => <option key={k} value={k}>{k}</option>)}
            </select>
            <input
              type="number"
              value={row.beiMauzo || ''}
              onChange={(e) => onChange({ beiMauzo: Number(e.target.value) })}
              placeholder="Bei mauzo"
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none"
            />
            <input
              type="number"
              value={row.beiUnunuzi || ''}
              onChange={(e) => onChange({ beiUnunuzi: Number(e.target.value) })}
              placeholder="Bei ununuzi"
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none"
            />
            <select
              value={row.uom}
              onChange={(e) => onChange({ uom: e.target.value })}
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none bg-white"
            >
              {UOM.map((u) => <option key={u} value={u}>{u}</option>)}
            </select>
            <input
              type="number"
              value={row.stoki || ''}
              onChange={(e) => onChange({ stoki: Number(e.target.value) })}
              placeholder="Stoki"
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none"
            />

            {/* MAHALI = SELECT YA MADUKA */}
            <select
              value={row.location}
              onChange={(e) => onChange({ location: e.target.value })}
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none bg-white"
            >
              <option value="">— Duka —</option>
              {maduka.map((d) => (
                <option key={d.id} value={d.jina}>
                  {d.jina.replace('Smart Hardware - ', '')}
                </option>
              ))}
            </select>

            <input
              type="date"
              value={row.expiry || ''}
              onChange={(e) => onChange({ expiry: e.target.value })}
              className="px-2.5 py-1.5 rounded-md border border-slate-200 text-xs focus:border-brand-500 outline-none"
            />
          </div>
        </div>
      )}
    </div>
  )
}

function emptyRow(defaultLocation = ''): BulkRow {
  return {
    _id: `row-${Math.random().toString(36).slice(2, 9)}`,
    jina: '',
    sku: '',
    kundi: 'Saruji',
    beiMauzo: 0,
    beiUnunuzi: 0,
    uom: 'Mfuko',
    stoki: 0,
    location: defaultLocation,
    expiry: undefined,
    iconKey: 'Saruji',
    pichaUrl: undefined,
    _skuManual: false,
  }
}

// ============================================================
// Historia
// ============================================================
function Historia({ uingizaji }: { uingizaji: UingizajiVifaa[] }) {
  const [q, setQ] = useState('')
  const [filter, setFilter] = useState<HaliUingizaji | 'zote'>('zote')

  const filtered = useMemo(() => {
    return uingizaji.filter((u) => {
      if (filter !== 'zote' && u.hali !== filter) return false
      const query = q.toLowerCase().trim()
      if (!query) return true
      return (
        u.id.toLowerCase().includes(query) ||
        u.dukaJina.toLowerCase().includes(query) ||

// Modal ya Malipo ya Deni

function MalipoDeniModal({
  mtejaId,
  selectedOrder,
  orders,
  onClose,
  onSuccess,
  ongezaMalipo,
}: {
  mtejaId: string
  selectedOrder: string | null
  orders: ReturnType<typeof useStore>['orders']
  onClose: () => void
  onSuccess: (msg: string, ok: boolean) => void
  ongezaMalipo: ReturnType<typeof useStore>['ongezaMalipo']
}) {
  const [njia, setNjia] = useState<NjiaMalipoDeni>('mpesa')
  const [kiasi, setKiasi] = useState(0)
  const [kumbukumbu, setKumbukumbu] = useState('')
  const [saving, setSaving] = useState(false)
  const [error, setError] = useState('')

  const order = selectedOrder ? orders.find((o) => o.id === selectedOrder) : null
  const jumlaBaki = order
    ? order.jumla -
      order.malipo
        .filter((m) => m.hali === 'imethibitishwa')
        .reduce((s, m) => s + m.kiasi, 0)
    : orders.reduce(
        (s, o) =>
          s +
          (o.jumla -
            o.malipo
              .filter((m) => m.hali === 'imethibitishwa')
              .reduce((a, m) => a + m.kiasi, 0)),
        0
      )

  const [initialized, setInitialized] = useState(false)
  if (!initialized && jumlaBaki > 0) {
    setKiasi(jumlaBaki)
    setInitialized(true)
  }

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    setError('')

    if (!mtejaId) return setError('Mteja haipo')
    if (kiasi <= 0) return setError('Kiasi lazima kiwe zaidi ya sifuri')
    if (kiasi > jumlaBaki) return setError('Kiasi hakiwezi kuzidi baki')

    if (njia !== 'cash' && !kumbukumbu.trim()) {
      return setError('Weka kumbukumbu ya malipo (reference)')
    }

    setSaving(true)

    setTimeout(() => {
      try {
        if (order) {
          ongezaMalipo({
            orderId: order.id,
            njia: njia as NjiaMalipo,
            simu: order.simu,
            kumbukumbu: njia === 'cash' ? 'CASH' : kumbukumbu.trim().toUpperCase(),
            kiasi,
            hali: njia === 'cash' ? 'imethibitishwa' : 'inasubiri',
          })
        } else {
          let remaining = kiasi
          const sorted = [...orders].sort((a, b) => {
            const aBaki =
              a.jumla -
              a.malipo
                .filter((m) => m.hali === 'imethibitishwa')
                .reduce((s, m) => s + m.kiasi, 0)
            const bBaki =
              b.jumla -
              b.malipo
                .filter((m) => m.hali === 'imethibitishwa')
                .reduce((s, m) => s + m.kiasi, 0)
            return aBaki - bBaki
          })

          for (const o of sorted) {
            if (remaining <= 0) break
            const bakiOrder =
              o.jumla -
              o.malipo
                .filter((m) => m.hali === 'imethibitishwa')
                .reduce((s, m) => s + m.kiasi, 0)
            if (bakiOrder <= 0) continue
            const malipoKiasi = Math.min(remaining, bakiOrder)
            ongezaMalipo({
              orderId: o.id,
              njia: njia as NjiaMalipo,
              simu: o.simu,
              kumbukumbu:
                njia === 'cash' ? 'CASH' : kumbukumbu.trim().toUpperCase(),
              kiasi: malipoKiasi,

function Fomu({ onClose, onSubmit }: { onClose: () => void; onSubmit: any }) {
  const [aina, setAina] = useState<AinaGharama>('mafuta')
  const [kiasi, setKiasi] = useState(0)
  const [maelezo, setMaelezo] = useState('')
  const [sababu, setSababu] = useState('')
  const [simu, setSimu] = useState('')
  const [error, setError] = useState('')

  const submit = (e: React.FormEvent) => {
    e.preventDefault()
    if (kiasi <= 0) return setError('Kiasi lazima kiwe zaidi ya 0')
    if (!maelezo.trim()) return setError('Maelezo yanahitajika')
    if (!sababu.trim()) return setError('Sababu inahitajika')
    if (!simu.trim()) return setError('Simu inahitajika')
    onSubmit({
      aina, kiasi, maelezo, sababu, simu,
      aliyeomba: 'Juma Hassan', duka: 'Kwembe',
    })
    onClose()
  }

  return (
    <div className="fixed inset-0 z-50 flex items-end sm:items-center justify-center p-0 sm:p-4">
      <div className="absolute inset-0 bg-black/40" onClick={onClose} />
      <div className="relative bg-white w-full sm:max-w-lg rounded-t-2xl sm:rounded-2xl shadow-2xl max-h-[92vh] overflow-y-auto">
        <div className="sticky top-0 bg-white border-b border-slate-200 px-4 py-3 flex justify-between items-center">
          <div className="font-semibold">Omba Matumizi</div>
          <button onClick={onClose} className="p-1.5 rounded-lg hover:bg-slate-100"><X size={18} /></button>
        </div>
        <form onSubmit={submit} className="p-4 space-y-3">
          <select value={aina} onChange={(e) => setAina(e.target.value as AinaGharama)} className="w-full px-3 py-2.5 rounded-lg border border-slate-200 outline-none text-sm">
            {AINA_GHARAMA.map((a) => <option key={a.key} value={a.key}>{a.label}</option>)}
          </select>
          <input type="number" value={kiasi || ''} onChange={(e) => setKiasi(Number(e.target.value))} placeholder="Kiasi" className="w-full px-3 py-2.5 rounded-lg border border-slate-200 outline-none text-sm" />
          <input value={maelezo} onChange={(e) => setMaelezo(e.target.value)} placeholder="Maelezo" className="w-full px-3 py-2.5 rounded-lg border border-slate-200 outline-none text-sm" />
          <input value={simu} onChange={(e) => setSimu(e.target.value)} placeholder="Simu" className="w-full px-3 py-2.5 rounded-lg border border-slate-200 outline-none text-sm" />
          <textarea value={sababu} onChange={(e) => setSababu(e.target.value)} placeholder="Sababu" rows={2} className="w-full px-3 py-2.5 rounded-lg border border-slate-200 outline-none text-sm resize-none" />
          {error && <div className="p-3 rounded-lg bg-red-50 text-red-700 text-sm">{error}</div>}
          <div className="flex gap-2 pt-2">
            <button type="button" onClick={onClose} className="px-4 py-2.5 rounded-lg bg-slate-100 text-sm">Ghairi</button>
            <button type="submit" className="flex-1 inline-flex items-center justify-center gap-1.5 px-4 py-2.5 rounded-lg bg-brand-600 text-white text-sm font-semibold"><Save size={16} /> Tuma</button>

export default function MuuzajiRisiti() {
  const { orders } = useStore()
  const [q, setQ] = useState('')

  const filtered = useMemo(() => {
    const query = q.toLowerCase().trim()
    return orders.filter((o) => {
      if (!query) return true
      return o.id.toLowerCase().includes(query) || o.mteja.toLowerCase().includes(query) || o.simu.includes(query)
    })
  }, [orders, q])

  const stats = useMemo(() => {
    const zote = orders.length
    const jumla = orders.reduce((s, o) => s + o.jumla, 0)
    const cash = orders
      .filter((o) => o.malipo.some((m) => m.njia === 'cash' && m.hali === 'imethibitishwa'))
      .reduce((s, o) => s + o.jumla, 0)
    const deni = orders.reduce((s, o) => {
      const jumlaIlipwa = o.malipo.filter((m) => m.hali === 'imethibitishwa').reduce((a, m) => a + m.kiasi, 0)
      return s + Math.max(o.jumla - jumlaIlipwa, 0)
    }, 0)
    return { zote, jumla, cash, deni }
  }, [orders])

  return (
    <div className="space-y-5">
      <h1 className="text-2xl md:text-3xl font-bold tracking-tight text-slate-900 flex items-center gap-2">
        <Receipt size={26} className="text-brand-600" /> Risiti
      </h1>

      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
        <Stat label="Risiti Zote" value={String(stats.zote)} Icon={Receipt} tone="text-blue-600" />
        <Stat label="Jumla ya Mauzo" value={formatTsh(stats.jumla)} Icon={CheckCircle2} tone="text-emerald-600" />

// ============================================================
// TABLE 1: Stoki
// ============================================================
function StokiTable({ bidhaa }: { bidhaa: ReturnType<typeof useStore>['bidhaa'] }) {
  const [q, setQ] = useState('')

  const filtered = useMemo(() => {
    const query = q.toLowerCase().trim()
    if (!query) return bidhaa
    return bidhaa.filter(
      (b) =>
        b.jina.toLowerCase().includes(query) ||
        b.sku.toLowerCase().includes(query) ||
        b.kundi.toLowerCase().includes(query)
    )
  }, [bidhaa, q])

  return (
    <div>
      <div className="p-3 border-b border-slate-100">
        <div className="relative">
          <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" />
          <input
            value={q}
            onChange={(e) => setQ(e.target.value)}
            placeholder="Tafuta bidhaa..."
            className="w-full pl-9 pr-3 py-2 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
      </div>

      {filtered.length === 0 ? (
        <div className="p-10 text-center">
          <Package size={32} className="mx-auto text-slate-300 mb-2" />
          <div className="text-sm text-slate-500">Hakuna bidhaa</div>
        </div>
      ) : (
        <div className="overflow-x-auto">
          <table className="w-full text-sm min-w-[760px]">
            <thead className="bg-slate-50 text-left text-slate-500">
              <tr>
                <th className="p-3 font-medium text-xs uppercase tracking-wide">Bidhaa</th>
                <th className="p-3 font-medium text-xs uppercase tracking-wide">Kundi</th>
                <th className="p-3 text-right font-medium text-xs uppercase tracking-wide">Stoki</th>
                <th className="p-3 text-right font-medium text-xs uppercase tracking-wide">Bei Mauzo</th>
                <th className="p-3 text-right font-medium text-xs uppercase tracking-wide">Bei Ununuzi</th>
                <th className="p-3 text-right font-medium text-xs uppercase tracking-wide">Thamani</th>
                <th className="p-3 font-medium text-xs uppercase tracking-wide">Expiry</th>
                <th className="p-3 font-medium text-xs uppercase tracking-wide">Hali</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-100">
              {filtered.map((b) => {
                const Icon = iconForKundi(b.iconKey)
                const hali = b.stoki <= 0 ? 'imeisha' : b.stoki <= 20 ? 'inaribia' : 'sawa'
                return (
                  <tr key={b.id} className="hover:bg-slate-50/50">
                    <td className="p-3">
                      <div className="flex items-center gap-2">
                        <div className="w-9 h-9 rounded-lg bg-slate-50 border border-slate-200 overflow-hidden grid place-items-center text-slate-600 shrink-0">
                          {b.pichaUrl ? (
                            <img src={b.pichaUrl} alt="" className="w-full h-full object-cover" />
                          ) : (
                            <Icon size={16} />
                          )}
                        </div>
                        <div className="min-w-0">
                          <div className="font-medium truncate">{b.jina}</div>
                          <div className="text-xs text-slate-500">{b.sku}</div>
                        </div>
                      </div>
                    </td>
                    <td className="p-3">
                      <span className="text-xs px-2 py-0.5 rounded bg-slate-100 text-slate-700">
                        {b.kundi}
                      </span>
                    </td>
                    <td className="p-3 text-right font-semibold">
                      {b.stoki} <span className="text-xs text-slate-500">{b.uom}</span>
                    </td>
                    <td className="p-3 text-right text-brand-700 font-semibold">
                      {formatTsh(b.beiMauzo)}
                    </td>
                    <td className="p-3 text-right text-slate-600">
                      {formatTsh(b.beiUnunuzi)}
                    </td>
                    <td className="p-3 text-right font-semibold">
                      {formatTsh(b.stoki * b.beiUnunuzi)}
                    </td>
                    <td className="p-3 text-slate-600 text-xs">
                      {b.expiry ?? '—'}
                    </td>
                    <td className="p-3">
                      <span
                        className={`inline-flex items-center gap-1 text-[10px] uppercase font-semibold px-1.5 py-0.5 rounded ${
                          hali === 'sawa'
                            ? 'text-emerald-700 bg-emerald-50'
                            : hali === 'inaribia'
                            ? 'text-amber-700 bg-amber-50'
                            : 'text-red-700 bg-red-50'
                        }`}
                      >
                        {hali === 'sawa' ? 'Sawa' : hali === 'inaribia' ? 'Inaribia' : 'Imeisha'}
                      </span>
                    </td>
                  </tr>
                )
              })}
            </tbody>
          </table>
        </div>
      )}
    </div>
  )
}

        {value}
      </div>
    </div>
  )
},

import { Plus, Search, X, Save, Phone, MapPin } from 'lucide-react'
import { useStore } from '../../store/StoreContext'
import { formatTsh } from '../../utils/faida'

function MtejaSummary() {
  const { wateja, orders, deniLaMteja } = useStore()
  const mteja = wateja[0]

  const stats = useMemo(() => {
    if (!mteja) return { deni: 0, orders: 0, manunuzi: 0 }
    const ordersZake = orders.filter((o) => o.simu === mteja.simu)
    const jumlaManunuzi = ordersZake.reduce((s, o) => s + o.jumla, 0)
    return {
      deni: deniLaMteja(mteja.id),
      orders: ordersZake.length,
      manunuzi: jumlaManunuzi,
    }
  }, [mteja, orders, deniLaMteja])

  return (
    <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
      <SumStat label="Deni Langu" value={formatTsh(stats.deni)} Icon={Wallet} tone={stats.deni > 0 ? 'text-red-600' : 'text-emerald-600'} />
      <SumStat label="Orders" value={String(stats.orders)} Icon={Receipt} tone="text-blue-600" />
      <SumStat label="Manunuzi Yote" value={formatTsh(stats.manunuzi)} Icon={TrendingUp} tone="text-emerald-600" />
    </div>
  )
}

function MuuzajiSummary() {
  const { wafanyakazi, orders, mauzo } = useStore()
  const muuzaji = wafanyakazi.find((w) => w.jina === 'Juma Hassan')
  const duka = muuzaji?.duka ?? 'Kwembe'

  const stats = useMemo(() => {
    const mauzoYake = mauzo.filter((m) => m.muuzaji === 'Juma Hassan')
    const jumlaMauzo = mauzoYake.reduce((s, m) => s + m.jumlaMauzo, 0)
    const jumlaFaida = mauzoYake.reduce((s, m) => s + m.faidaGhafi, 0)
    const ordersZake = orders.length
    return { jumlaMauzo, jumlaFaida, ordersZake }
  }, [mauzo, orders])

  return (
    <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
      <SumStat label="Duka" value={duka} Icon={Store} tone="text-purple-600" />
      <SumStat label="Mauzo Yangu" value={formatTsh(stats.jumlaMauzo)} Icon={Receipt} tone="text-blue-600" />
      <SumStat label="Faida Ghafi" value={formatTsh(stats.jumlaFaida)} Icon={TrendingUp} tone="text-emerald-600" />
    </div>
  )
}

function MmilikiSummary() {

  },
]

function loadSettings(role: Role): Settings {
  try {
    const raw = localStorage.getItem(`sh_settings_${role}`)
    if (!raw) return { ...DEFAULTS }
    return { ...DEFAULTS, ...JSON.parse(raw) }
  } catch {
    return { ...DEFAULTS }
  }
}

function saveSettings(role: Role, settings: Settings) {
  try {
    localStorage.setItem(`sh_settings_${role}`, JSON.stringify(settings))
    return true
  } catch {
    return false
  }
}

function ChangePasswordModal({
  onClose,
  onSuccess,
}: {
  onClose: () => void
  onSuccess: () => void
}) {
  const [current, setCurrent] = useState('')
  const [password, setPassword] = useState('')
  const [confirm, setConfirm] = useState('')
  const [showPass, setShowPass] = useState(false)
  const [saving, setSaving] = useState(false)
  const [error, setError] = useState('')

  const submit = (e: React.FormEvent) => {
    e.preventDefault()
    if (!current.trim()) return setError('Nywila ya sasa inahitajika')
    if (password.length < 6) return setError('Nywila mpya iwe angalau herufi 6')
    if (password !== confirm) return setError('Nywila hazifanani')

    setError('')
    setSaving(true)
    setTimeout(() => {
      setSaving(false)
      onSuccess()
    }, 500)
  }

  return (
    <div className="fixed inset-0 z-50 flex items-end sm:items-center justify-center p-0 sm:p-4">
      <div className="absolute inset-0 bg-black/40" onClick={() => !saving && onClose()} />
      <div className="relative bg-white w-full sm:max-w-md rounded-t-2xl sm:rounded-2xl shadow-2xl">

            </div>
            <div className="text-xs text-slate-500">

            </div>
          </div>
          <button
            onClick={() => !saving && onClose()}
            className="p-1.5 rounded-lg hover:bg-slate-100"
            disabled={saving}
          >
            <X size={18} />
          </button>
        </div>

        <form onSubmit={submit} className="p-4 space-y-3">
          <div>
            <label className="text-sm font-medium">Nywila ya Sasa</label>
            <input
              type={showPass ? 'text' : 'password'}
              value={current}
              onChange={(e) => setCurrent(e.target.value)}
              className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
              placeholder="••••••••"

            <label className="text-sm font-medium">Nywila Mpya</label>
            <div className="relative mt-1">
              <input
                type={showPass ? 'text' : 'password'}
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                className="w-full px-3 py-2.5 pr-10 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"

              >
                {showPass ? <EyeOff size={14} /> : <Eye size={14} />}
              </button>
            </div>
          </div>
          <div>
            <label className="text-sm font-medium">Rudia Nywila Mpya</label>
            <input
              type={showPass ? 'text' : 'password'}
              value={confirm}
              onChange={(e) => setConfirm(e.target.value)}
              className="mt-1 w-full px-3 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
              placeholder="••••••••"
            />
          </div>

          {error && (
            <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 text-red-700 text-sm">
              <AlertTriangle size={16} className="mt-0.5 shrink-0" />
              {error}
            </div>
          )}

            <button

        </form>

              className="inline-flex items-center gap-1.5 px-3 py-2 bg-brand-50 text-brand-700 rounded-lg text-sm font-semibold hover:bg-brand-100"
            >
              <CheckCheck size={14} />
              Soma Zote
            </button>
          )}
          {notifications.length > 0 && (
            <button
              onClick={futaZote}
              className="inline-flex items-center gap-1.5 px-3 py-2 bg-red-50 text-red-700 rounded-lg text-sm font-semibold hover:bg-red-100"
            >
              <Trash2 size={14} />
              Futa Zote
            </button>
          )}
        </div>
      </div>

      {/* Filters */}
      <div className="bg-white rounded-xl border border-slate-200 p-3 md:p-4 space-y-3">
        <div className="relative">
          <Search
            size={18}
            className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400 pointer-events-none"
          />
          <input
            value={q}
            onChange={(e) => setQ(e.target.value)}
            placeholder="Tafuta taarifa..."
            className="w-full pl-10 pr-4 py-2.5 rounded-lg border border-slate-200 focus:border-brand-500 outline-none text-sm"
          />
        </div>
        <div className="flex flex-wrap gap-2 items-center">
          <Filter size={14} className="text-slate-400" />
          {FILTERS.map((f) => (
            <button
              key={f.key}
              onClick={() => setFilter(f.key)}
              className={`px-3 py-1.5 rounded-lg text-sm border transition ${
                filter === f.key
                  ? 'bg-brand-600 text-white border-brand-600'
                  : 'bg-white border-slate-200 hover:bg-slate-50'
              }`}
            >
              {f.label}
            </button>
          ))}
          <button
            onClick={() => setShowOnlyUnread((v) => !v)}
            className={`ml-auto px-3 py-1.5 rounded-lg text-sm border transition ${
              showOnlyUnread
                ? 'bg-brand-600 text-white border-brand-600'
                : 'bg-white border-slate-200 hover:bg-slate-50'
            }`}
          >
            Hazijasomwa ({unreadCount})
          </button>
        </div>
      </div>

      {/* List */}
      {filtered.length === 0 ? (
        <div className="bg-white rounded-xl border border-slate-200 p-10 text-center">
          <Inbox size={40} className="mx-auto mb-3 text-slate-300" />
          <div className="font-semibold">
            {notifications.length === 0
              ? 'Hakuna taarifa'
              : 'Hakuna taarifa inayolingana'}
          </div>
          <p className="text-sm text-slate-500 mt-1">
            {notifications.length === 0
              ? 'Ukipata taarifa mpya, zitaonekana hapa.'
              : 'Badilisha kichujio au utafutaji.'}
          </p>
        </div>
      ) : (
        <div className="space-y-2">
          {filtered.map((n) => {
            const meta = AINA_META[n.type]
            const kundiMeta = KUNDI_META[n.kundi]
            return (
              <div
                key={n.id}
                className={`bg-white rounded-xl border p-4 transition ${
                  !n.read
                    ? 'border-brand-200 bg-brand-50/30'
                    : 'border-slate-200'
                }`}
              >
                <div className="flex items-start gap-3">
                  <div
                    className={`w-10 h-10 rounded-lg grid place-items-center shrink-0 border ${meta.color}`}
                  >
                    <meta.Icon size={18} />
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center gap-2 flex-wrap">
                      <button
                        onClick={() => handleClick(n)}
                        className="font-semibold text-sm text-left hover:text-brand-600"
                      >
                        {n.title}
                      </button>
                      {!n.read && (
                        <span className="w-2 h-2 rounded-full bg-brand-600 shrink-0" />
                      )}
                      <span className="text-[10px] uppercase font-semibold px-1.5 py-0.5 rounded bg-slate-100 text-slate-600">
                        {kundiMeta.label}
                      </span>
                    </div>
                    <div className="text-sm text-slate-600 mt-0.5">
                      {n.message}
                    </div>
                    <div className="text-xs text-slate-400 mt-1">{n.time}</div>
                  </div>
                  <div className="flex items-center gap-1 shrink-0">
                    <button
                      onClick={() => handleClick(n)}
                      className="p-2 rounded-lg text-brand-600 hover:bg-brand-50"
                      title="Fungua"
                    >
                      <ArrowRight size={16} />
                    </button>
                    <button
                      onClick={() => futa(n.id)}
                      className="p-2 rounded-lg text-red-500 hover:bg-red-50"
                      title="Futa"
                    >
                      <Trash2 size={16} />
                    </button>
                  </div>
                </div>
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
},
